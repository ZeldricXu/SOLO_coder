# GameStats InfluxDB 初始化配置
# 创建组织、存储桶和令牌

# 1. 创建组织
influx org create --name gamestats --description "GameStats Analytics Organization"

# 2. 创建存储桶
influx bucket create --name events --org gamestats --retention 90d --description "Game events data (90 days retention)"
influx bucket create --name stats --org gamestats --retention 365d --description "Statistics data (365 days retention)"
influx bucket create --name monitoring --org gamestats --retention 7d --description "System monitoring data (7 days retention)"

# 3. 创建API令牌 (保存输出的token)
influx auth create --org gamestats --all-access --description "Full access token for GameStats"
